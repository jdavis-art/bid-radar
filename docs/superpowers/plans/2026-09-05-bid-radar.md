# Bid Radar Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A local, scheduled Python pipeline that sweeps BidNet Direct, PublicPurchase, and SAM.gov for housing solicitations, normalizes and dedupes them into SQLite, scores them from a YAML rule set, and writes a daily digest.

**Architecture:** Source adapters yield raw dicts behind one `Source` protocol; `normalize` turns them into `Solicitation` records; `scoring` is a config-driven rule engine that must reproduce the legacy scorer exactly; `store` is a small SQLite layer; `digest` renders markdown and JSON. A thin argparse CLI ties them together and a PowerShell script runs it from Task Scheduler.

**Tech Stack:** Python 3.12+ (`py` launcher), `requests`, `lxml`, `pyyaml`, `pytest`. No async, no browser automation, no framework.

**Spec:** `docs/superpowers/specs/2026-09-05-bid-radar-design.md`

## Global Constraints

- Python 3.12 or newer, invoked as `py` on this machine. Create the venv with `py -m venv .venv` and use `.venv\Scripts\python.exe` for every run.
- Dependencies limited to `requests`, `lxml`, `pyyaml` at runtime and `pytest` for tests. Nothing else without a spec change.
- Cookie files live at `~/.bidnet-curl.txt` and `~/.publicpurchase-curl.txt`. They are never read into logs, never committed, and every error message that mentions them names the path only.
- `config.local.yaml`, `data/`, `logs/`, `seed/`, `digest/`, `.env` are gitignored (already in `.gitignore`). Never commit anything from those paths.
- The scoring regression must reproduce the 2026-09-04 distribution exactly: 4,073 at 0, 77 at 1, 2 at 2, 67 at 3, 14 at 4, 4 at 5.
- A sweep never aborts because one source fails; per-source failures are recorded in the `runs` table.
- Commit after every task with the message shown. Git identity is already set for this repo.
- **Gate rule:** at Gates 1 to 4 (marked below) STOP, print the requested output, and wait for the supervising session's green light before continuing.

```python
"""Configuration: config.yaml (public defaults) + config.local.yaml (private overlay) + env."""
from __future__ import annotations

import copy
import os
from pathlib import Path

import yaml

ENV_OUTPUT_DIR = "BID_RADAR_OUTPUT_DIR"
ENV_SAM_KEY = "SAM_API_KEY"


def _deep_merge(base: dict, overlay: dict) -> dict:
    out = copy.deepcopy(base)
    for key, value in overlay.items():
        if isinstance(value, dict) and isinstance(out.get(key), dict):
            out[key] = _deep_merge(out[key], value)
        else:
            out[key] = copy.deepcopy(value)
    return out


def project_root() -> Path:
    return Path(__file__).resolve().parent.parent


def load_config(root: Path | None = None) -> dict:
    root = Path(root) if root else project_root()
    with open(root / "config.yaml", encoding="utf-8") as fh:
        cfg = yaml.safe_load(fh) or {}
    local = root / "config.local.yaml"
    if local.exists():
        with open(local, encoding="utf-8") as fh:
            cfg = _deep_merge(cfg, yaml.safe_load(fh) or {})
    if os.environ.get(ENV_OUTPUT_DIR):
        cfg["output_dir"] = os.environ[ENV_OUTPUT_DIR]
    if os.environ.get(ENV_SAM_KEY):
        cfg.setdefault("sources", {}).setdefault("samgov", {})["api_key"] = os.environ[ENV_SAM_KEY]
    cfg["_root"] = str(root)
    return cfg


def resolve_path(cfg: dict, key: str) -> Path:
    """Resolve output_dir / data_dir / log_dir relative to the project root, expanding ~."""
    raw = Path(os.path.expanduser(cfg[key]))
    return raw if raw.is_absolute() else Path(cfg["_root"]) / raw
```

```python
import dataclasses
import datetime as dt
import json
import re

STATES = {
    "AL": "Alabama", "AK": "Alaska", "AZ": "Arizona", "AR": "Arkansas", "CA": "California",
    "CO": "Colorado", "CT": "Connecticut", "DE": "Delaware", "FL": "Florida", "GA": "Georgia",
    "HI": "Hawaii", "ID": "Idaho", "IL": "Illinois", "IN": "Indiana", "IA": "Iowa",
    "KS": "Kansas", "KY": "Kentucky", "LA": "Louisiana", "ME": "Maine", "MD": "Maryland",
    "MA": "Massachusetts", "MI": "Michigan", "MN": "Minnesota", "MS": "Mississippi",
    "MO": "Missouri", "MT": "Montana", "NE": "Nebraska", "NV": "Nevada", "NH": "New Hampshire",
    "NJ": "New Jersey", "NM": "New Mexico", "NY": "New York", "NC": "North Carolina",
    "ND": "North Dakota", "OH": "Ohio", "OK": "Oklahoma", "OR": "Oregon", "PA": "Pennsylvania",
    "RI": "Rhode Island", "SC": "South Carolina", "SD": "South Dakota", "TN": "Tennessee",
    "TX": "Texas", "UT": "Utah", "VT": "Vermont", "VA": "Virginia", "WA": "Washington",
    "WV": "West Virginia", "WI": "Wisconsin", "WY": "Wyoming", "DC": "District of Columbia",
}

_DATE_FORMATS = ("%m/%d/%Y", "%Y-%m-%d", "%m/%d/%y")
_DATETIME_FORMATS = ("%m/%d/%Y %I:%M %p", "%m/%d/%Y %H:%M", "%Y-%m-%dT%H:%M:%S")


def normalize_state(text: str) -> str:
    t = (text or "").strip()
    if not t:
        return ""
    if t.upper() in STATES:
        return STATES[t.upper()]
    return t


def parse_date(text: str | None) -> dt.date | None:
    t = (text or "").strip()
    if not t:
        return None
    head = t[:10]
    for fmt in _DATE_FORMATS:
        try:
            return dt.datetime.strptime(head, fmt).date()
        except ValueError:
            continue
    m = re.match(r"(\d{1,2})/(\d{1,2})/(\d{4})", t)
    if m:
        return dt.date(int(m.group(3)), int(m.group(1)), int(m.group(2)))
    return None


def parse_datetime(text: str | None) -> dt.datetime | None:
    t = (text or "").strip()
    if not t:
        return None
    # Drop a trailing timezone token such as "EDT" or "MST"; times are kept naive.
    t = re.sub(r"\s+[A-Z]{2,4}$", "", t)
    if "T" in t:
        base = t[:19]
        try:
            return dt.datetime.strptime(base, "%Y-%m-%dT%H:%M:%S")
        except ValueError:
            pass
    for fmt in _DATETIME_FORMATS:
        try:
            return dt.datetime.strptime(t, fmt)
        except ValueError:
            continue
    d = parse_date(t)
    return dt.datetime(d.year, d.month, d.day) if d else None


@dataclasses.dataclass
class Solicitation:
    uid: str
    source: str
    source_id: str
    title: str
    agency: str = ""
    location: str = ""
    bid_type: str = ""
    published: dt.date | None = None
    closing: dt.datetime | None = None
    url: str = ""
    blurb: str = ""
    mandatory_prebid: bool = False
    extra: dict = dataclasses.field(default_factory=dict)

    def to_row(self) -> dict:
        row = dataclasses.asdict(self)
        row["published"] = self.published.isoformat() if self.published else None
        row["closing"] = self.closing.isoformat(timespec="minutes") if self.closing else None
        row["mandatory_prebid"] = int(self.mandatory_prebid)
        row["extra"] = json.dumps(self.extra, sort_keys=True)
        return row

    @classmethod
    def from_row(cls, row: dict) -> "Solicitation":
        return cls(
            uid=row["uid"], source=row["source"], source_id=row["source_id"], title=row["title"],
            agency=row.get("agency") or "", location=row.get("location") or "",
            bid_type=row.get("bid_type") or "",
            published=dt.date.fromisoformat(row["published"]) if row.get("published") else None,
            closing=dt.datetime.fromisoformat(row["closing"]) if row.get("closing") else None,
            url=row.get("url") or "", blurb=row.get("blurb") or "",
            mandatory_prebid=bool(row.get("mandatory_prebid")),
            extra=json.loads(row["extra"]) if isinstance(row.get("extra"), str) else (row.get("extra") or {}),
        )
```

```python
"""Per-source raw dict -> Solicitation."""
from __future__ import annotations

from .model import Solicitation, normalize_state, parse_date, parse_datetime


def _bidnet(raw: dict) -> Solicitation:
    extra = {"searches": raw.get("searches", [])}
    closing = parse_datetime(raw.get("closing"))
    if raw.get("closing") and closing is None:
        extra["closing_raw"] = raw["closing"]
    published = parse_date(raw.get("published"))
    if raw.get("published") and published is None:
        extra["published_raw"] = raw["published"]
    return Solicitation(
        uid=f"bidnet:{raw['id']}", source="bidnet", source_id=str(raw["id"]),
        title=raw.get("title", "").strip(), agency=raw.get("agency", "").strip(),
        location=normalize_state(raw.get("location", "")), bid_type=raw.get("bid_type", ""),
        published=published, closing=closing, url=raw.get("url", ""),
        blurb=raw.get("blurb", ""), mandatory_prebid=bool(raw.get("mandatory_prebid")), extra=extra,
    )


def _samgov(raw: dict) -> Solicitation:
    pop = raw.get("placeOfPerformance") or {}
    state = ((pop.get("state") or {}).get("code")) or ""
    extra = {"type": raw.get("type", ""), "naics": raw.get("naicsCode", ""),
             "set_aside": raw.get("typeOfSetAsideDescription", "")}
    closing = parse_datetime(raw.get("responseDeadLine"))
    if raw.get("responseDeadLine") and closing is None:
        extra["closing_raw"] = raw["responseDeadLine"]
    return Solicitation(
        uid=f"samgov:{raw['noticeId']}", source="samgov", source_id=str(raw["noticeId"]),
        title=raw.get("title", "").strip(), agency=(raw.get("fullParentPathName") or "").strip(),
        location=normalize_state(state), bid_type="Federal",
        published=parse_date(raw.get("postedDate")), closing=closing,
        url=raw.get("uiLink", ""), blurb=raw.get("description_excerpt", ""),
        mandatory_prebid=False, extra=extra,
    )


def _publicpurchase(raw: dict) -> Solicitation:
    extra = {"agency_slug": raw.get("agency_slug", "")}
    closing = parse_datetime(raw.get("closing"))
    if raw.get("closing") and closing is None:
        extra["closing_raw"] = raw["closing"]
    return Solicitation(
        uid=f"publicpurchase:{raw['id']}", source="publicpurchase", source_id=str(raw["id"]),
        title=raw.get("title", "").strip(), agency=raw.get("agency", "").strip(),
        location=normalize_state(raw.get("location", "")), bid_type="State & Local Bids",
        published=parse_date(raw.get("published")), closing=closing,
        url=raw.get("url", ""), blurb=raw.get("blurb", ""), mandatory_prebid=False, extra=extra,
    )


_NORMALIZERS = {"bidnet": _bidnet, "samgov": _samgov, "publicpurchase": _publicpurchase}


def normalize(source: str, raw: dict) -> Solicitation:
    try:
        fn = _NORMALIZERS[source]
    except KeyError:
        raise ValueError(f"unknown source {source!r}") from None
    return fn(raw)
```

```python
"""Config-driven scorer. Mirrors the legacy scorer's order of checks exactly:
exclude (title) -> not-housing (title) -> housing (title, else blurb) -> location -> boost -> federal note."""
from __future__ import annotations

import dataclasses
import re

from .model import Solicitation


def _compile(patterns: list[str]) -> re.Pattern | None:
    patterns = [p for p in (patterns or []) if p]
    if not patterns:
        return None
    return re.compile("|".join(f"(?:{p})" for p in patterns), re.I)


@dataclasses.dataclass
class Rules:
    exclude_title: re.Pattern | None
    not_housing_title: re.Pattern | None
    housing: re.Pattern | None
    location_points: dict[str, int]
    boost: re.Pattern | None
    federal_bid_type: str
    federal_note: str
    digest_min_score: int = 3

    @classmethod
    def from_config(cls, sc: dict) -> "Rules":
        return cls(
            exclude_title=_compile(sc.get("exclude_title")),
            not_housing_title=_compile(sc.get("not_housing_title")),
            housing=_compile(sc.get("housing")),
            location_points={k: int(v) for k, v in (sc.get("location_points") or {}).items()},
            boost=_compile(sc.get("boost")),
            federal_bid_type=sc.get("federal_bid_type", "Federal Bids"),
            federal_note=sc.get("federal_note", "federal (CAGE/SAM required)"),
            digest_min_score=int(sc.get("digest_min_score", 3)),
        )


def score(sol: Solicitation, rules: Rules) -> tuple[int, str]:
    title = sol.title or ""
    text = title + " " + (sol.blurb or "")
    if rules.exclude_title and (m := rules.exclude_title.search(title)):
        return 0, "excluded: " + m.group(0).lower()
    if rules.not_housing_title and (m := rules.not_housing_title.search(title)):
        return 0, "not housing (" + m.group(0).lower() + ")"
    if not rules.housing or not rules.housing.search(title):
        if rules.housing and rules.housing.search(text):
            return 1, "housing word in blurb only"
        return 0, "not housing"
    sc = 2 + rules.location_points.get(sol.location, 0)
    if rules.boost and rules.boost.search(text):
        sc += 1
    if sol.bid_type == rules.federal_bid_type:
        return sc, rules.federal_note
    return sc, ""
```

```python
"""SQLite persistence. One row per uid; nothing is ever deleted."""
from __future__ import annotations

import datetime as dt
import json
import sqlite3
from pathlib import Path

from .model import Solicitation

SCHEMA = """
create table if not exists solicitations (
  uid text primary key, source text not null, source_id text not null, title text not null,
  agency text, location text, bid_type text, published text, closing text, url text, blurb text,
  mandatory_prebid integer default 0, extra text default '{}',
  score integer default 0, score_note text default '', status text default 'new',
  first_seen text not null, last_seen text not null
);
create index if not exists ix_sol_score on solicitations(score);
create index if not exists ix_sol_closing on solicitations(closing);
create table if not exists duplicates (
  uid_a text not null, uid_b text not null, reason text, primary key (uid_a, uid_b)
);
create table if not exists runs (
  id integer primary key autoincrement, started text not null, finished text,
  per_source text default '{}', error text
);
"""

MUTABLE = ("title", "agency", "location", "bid_type", "published", "closing", "url", "blurb",
           "mandatory_prebid", "extra", "score", "score_note")


def _ts(t: dt.datetime) -> str:
    return t.isoformat(timespec="minutes")


class Store:
    def __init__(self, path: Path | str):
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.db = sqlite3.connect(str(self.path))
        self.db.row_factory = sqlite3.Row
        self.db.executescript(SCHEMA)

    def close(self):
        self.db.close()

    # ----------------------------------------------------------- solicitations
    def upsert(self, sol: Solicitation, score: int, note: str, seen_at: dt.datetime) -> bool:
        row = sol.to_row()
        row["score"], row["score_note"] = score, note
        existing = self.db.execute("select uid from solicitations where uid=?", (sol.uid,)).fetchone()
        if existing:
            sets = ", ".join(f"{c}=?" for c in MUTABLE)
            self.db.execute(f"update solicitations set {sets}, last_seen=? where uid=?",
                            [row[c] for c in MUTABLE] + [_ts(seen_at), sol.uid])
            self.db.commit()
            return False
        cols = ["uid", "source", "source_id"] + list(MUTABLE) + ["first_seen", "last_seen"]
        vals = [row["uid"], row["source"], row["source_id"]] + [row[c] for c in MUTABLE] + [_ts(seen_at), _ts(seen_at)]
        self.db.execute(f"insert into solicitations ({', '.join(cols)}) values ({', '.join('?' * len(cols))})", vals)
        self.db.commit()
        return True

    def get_row(self, uid: str) -> dict | None:
        r = self.db.execute("select * from solicitations where uid=?", (uid,)).fetchone()
        return dict(r) if r else None

    def get(self, uid: str) -> Solicitation | None:
        r = self.get_row(uid)
        return Solicitation.from_row(r) if r else None

    def open_rows(self, min_score: int, as_of: dt.datetime) -> list[dict]:
        rows = self.db.execute(
            "select * from solicitations where score>=? and status in ('new','seen','pursued') "
            "and (closing is null or closing>=?) order by score desc, closing asc, title asc",
            (min_score, _ts(as_of))).fetchall()
        return [dict(r) for r in rows]

    def new_since(self, ts: dt.datetime, min_score: int) -> list[dict]:
        rows = self.db.execute(
            "select * from solicitations where score>=? and first_seen>? and status<>'dismissed' "
            "order by score desc, closing asc, title asc", (min_score, _ts(ts))).fetchall()
        return [dict(r) for r in rows]

    def mark(self, uid: str, status: str):
        if status not in ("new", "seen", "dismissed", "pursued"):
            raise ValueError(f"bad status {status!r}")
        self.db.execute("update solicitations set status=? where uid=?", (status, uid))
        self.db.commit()

    def all_rows(self) -> list[dict]:
        return [dict(r) for r in self.db.execute("select * from solicitations").fetchall()]

    # ------------------------------------------------------------------ runs
    def start_run(self) -> int:
        cur = self.db.execute("insert into runs (started) values (?)", (_ts(dt.datetime.now()),))
        self.db.commit()
        return cur.lastrowid

    def finish_run(self, run_id: int, per_source: dict, error: str | None):
        self.db.execute("update runs set finished=?, per_source=?, error=? where id=?",
                        (_ts(dt.datetime.now()), json.dumps(per_source, sort_keys=True), error, run_id))
        self.db.commit()

    def last_run(self) -> dict | None:
        r = self.db.execute("select * from runs order by id desc limit 1").fetchone()
        return dict(r) if r else None

    def last_successful_run(self) -> dt.datetime | None:
        r = self.db.execute("select started from runs where finished is not null and error is null "
                            "order by id desc limit 1").fetchone()
        return dt.datetime.fromisoformat(r["started"]) if r else None

    # ------------------------------------------------------------ duplicates
    def link_duplicate(self, uid_a: str, uid_b: str, reason: str):
        a, b = sorted((uid_a, uid_b))
        self.db.execute("insert or ignore into duplicates (uid_a, uid_b, reason) values (?,?,?)", (a, b, reason))
        self.db.commit()

    def duplicates_of(self, uid: str) -> list[str]:
        rows = self.db.execute("select uid_a, uid_b from duplicates where uid_a=? or uid_b=?", (uid, uid)).fetchall()
        return sorted(r["uid_b"] if r["uid_a"] == uid else r["uid_a"] for r in rows)
```

```python
"""Cross-source duplicate linking. Same uid is handled by Store.upsert; this links different sources."""
from __future__ import annotations

import datetime as dt
import itertools
import re

from .store import Store

_AGENCY_STOP = {"city", "of", "county", "town", "the", "authority", "housing", "department", "dept"}
_MAX_DAYS = 3


def title_key(title: str, agency: str) -> str:
    agency_tokens = {t for t in re.findall(r"[a-z0-9]+", (agency or "").lower()) if t not in _AGENCY_STOP}
    words = re.findall(r"[a-z0-9]+", (title or "").lower())
    return " ".join(w for w in words if w not in agency_tokens)


def _closing(row: dict) -> dt.datetime | None:
    return dt.datetime.fromisoformat(row["closing"]) if row.get("closing") else None


def link_duplicates(store: Store, rows: list[dict]) -> int:
    groups: dict[str, list[dict]] = {}
    for r in rows:
        groups.setdefault(title_key(r["title"], r.get("agency", "")), []).append(r)
    new_links = 0
    for key, members in groups.items():
        if not key or len(members) < 2:
            continue
        for a, b in itertools.combinations(members, 2):
            if a["source"] == b["source"]:
                continue
            ca, cb = _closing(a), _closing(b)
            if ca is None or cb is None or abs((ca - cb).days) > _MAX_DAYS:
                continue
            before = len(store.duplicates_of(a["uid"]))
            store.link_duplicate(a["uid"], b["uid"], "title+closing")
            if len(store.duplicates_of(a["uid"])) > before:
                new_links += 1
    return new_links
```

```python
"""Daily digest: new-since-last-run first, then everything open at or above the threshold."""
from __future__ import annotations

import datetime as dt
import json
from pathlib import Path

from .scoring import Rules
from .store import Store

_COLS = ("score", "location", "title", "agency", "closing", "source", "url", "score_note")


def _slim(store: Store, row: dict) -> dict:
    out = {c: row.get(c) for c in ("uid",) + _COLS}
    out["duplicates"] = store.duplicates_of(row["uid"])
    return out


def build_digest(store: Store, rules: Rules, as_of: dt.datetime, since: dt.datetime | None) -> dict:
    min_score = rules.digest_min_score
    new = store.new_since(since, min_score) if since else store.open_rows(min_score, as_of)
    new_uids = {r["uid"] for r in new}
    open_rest = [r for r in store.open_rows(min_score, as_of) if r["uid"] not in new_uids]
    low = [r for r in store.open_rows(1, as_of) if r["score"] < min_score]
    last = store.last_run() or {}
    sources = json.loads(last.get("per_source") or "{}")
    return {
        "as_of": as_of.isoformat(timespec="minutes"),
        "since": since.isoformat(timespec="minutes") if since else None,
        "sources": sources,
        "run_error": last.get("error"),
        "new": [_slim(store, r) for r in new],
        "open": [_slim(store, r) for r in open_rest],
        "low_count": len(low),
    }


def _table(rows: list[dict]) -> str:
    if not rows:
        return "_none_\n"
    lines = ["| Score | State | Title | Agency | Closing | Source | Link | Note |",
             "| --- | --- | --- | --- | --- | --- | --- | --- |"]
    for r in rows:
        dup = f" (+{len(r['duplicates'])} dup)" if r.get("duplicates") else ""
        lines.append(f"| {r['score']} | {r['location'] or ''} | {r['title']} | {r['agency'] or ''} | "
                     f"{(r['closing'] or '')[:16]} | {r['source']}{dup} | {r['url'] or ''} | {r['score_note'] or ''} |")
    return "\n".join(lines) + "\n"


def render_markdown(d: dict) -> str:
    day = d["as_of"][:10]
    out = [f"# Bid Radar digest {day}", ""]
    out.append("## Run")
    for name, info in sorted(d["sources"].items()):
        out.append(f"- {name}: {info.get('rows', 0)} rows, {info.get('health', '?')}")
    if d.get("run_error"):
        out.append(f"- error: {d['run_error']}")
    if not d["sources"]:
        out.append("- no run recorded")
    out.append("")
    out.append(f"## New since {d['since'] or 'the beginning'}")
    out.append(_table(d["new"]))
    out.append("## Open, score 3 and up")
    out.append(_table(d["open"]))
    out.append(f"## Below threshold: {d['low_count']} open rows (see the JSON)")
    out.append("")
    return "\n".join(out)


def write_digest(store: Store, rules: Rules, output_dir: Path, as_of: dt.datetime,
                 since: dt.datetime | None) -> tuple[Path, Path]:
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)
    d = build_digest(store, rules, as_of, since)
    stem = f"bid-radar-{as_of.date().isoformat()}"
    md_path, json_path = output_dir / f"{stem}.md", output_dir / f"{stem}.json"
    md_path.write_text(render_markdown(d), encoding="utf-8")
    json_path.write_text(json.dumps(d, indent=1, ensure_ascii=False), encoding="utf-8")
    return md_path, json_path
```

```python
"""Parse a 'Copy as cURL (bash)' file into a Cookie header. Values never appear in logs."""
from __future__ import annotations

import os
import re
from pathlib import Path


class CookieFileMissing(FileNotFoundError):
    pass


class CookieNotFound(ValueError):
    pass


HANDOFF = ("In the logged-in Chrome tab: F12 -> Network -> reload -> right-click any request "
           "for that site -> Copy -> Copy as cURL (bash) -> paste into {path}")


def cookie_header_from_curl_file(path: Path | str) -> str:
    path = Path(os.path.expanduser(str(path)))
    if not path.exists():
        raise CookieFileMissing(f"{path} not found. " + HANDOFF.format(path=path))
    raw = path.read_text(encoding="utf-8", errors="replace")
    m = re.search(r"-H\s+(['\"])cookie:\s*(.*?)\1", raw, re.I | re.S)
    if m:
        return m.group(2).strip()
    m = re.search(r"(?:-b|--cookie)\s+(['\"])(.*?)\1", raw, re.S)
    if m:
        return m.group(2).strip()
    m = re.search(r"^\s*cookie:\s*(.+)$", raw, re.I | re.M)
    if m:
        return m.group(1).strip()
    stripped = raw.strip()
    if "=" in stripped and ";" in stripped and "\n" not in stripped:
        return stripped
    raise CookieNotFound(f"No Cookie header found in {path}. " + HANDOFF.format(path=path))


def redact(text: str) -> str:
    return re.sub(r"((?:set-)?cookie:\s*).*", r"\1[redacted]", text or "", flags=re.I)
```

```python
"""BidNet Direct adapter. Verified mechanics (2026-09-04):
- explicit pageNumber/pageSize/selectedSavedSearch on every request (the server remembers pageNumber);
- rows are tr.mets-table-row; strip <script> before reading text;
- native abstracts live at /private/solicitations/<id>/abstract (GET without the query string);
- aggregated notices live at /private/supplier/solicitations/notice/<id>/abstract;
- federal notices 404 everywhere here (read them from SAM.gov)."""
from __future__ import annotations

import re
import time
from typing import Iterator

import requests
from lxml import html as lxhtml

from ..cookies import cookie_header_from_curl_file, CookieFileMissing, CookieNotFound
from .base import NeedsCookie, HEALTH_OK, HEALTH_NEEDS_COOKIE, HEALTH_DOWN

UA = ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
      "(KHTML, like Gecko) Chrome/124.0 Safari/537.36")


class FederalNotice(Exception):
    pass


def strip_scripts(d):
    for el in d.xpath("//script|//style"):
        el.getparent().remove(el)
    return d


def text(el) -> str:
    return re.sub(r"\s+", " ", el.text_content()).strip() if el is not None else ""


def first(tr, cls) -> str:
    hit = tr.xpath(f".//*[contains(@class,'{cls}')]")
    return text(hit[0]) if hit else ""


def parse_rows(d, base_url: str = "https://www.bidnetdirect.com") -> list[dict]:
    rows = []
    for tr in d.xpath("//tr[contains(@class,'mets-table-row')]"):
        a = tr.xpath(".//*[contains(@class,'solicitationTitle')]//a")
        if not a:
            continue
        a = a[0]
        href = (a.get("href") or "").split("?")[0]
        sid = href.rstrip("/").split("/")[-1]
        t = text(tr)
        rows.append({
            "id": sid,
            "title": text(a),
            "agency": first(tr, "buyerIdentification"),
            "bid_type": first(tr, "searchContentGroupName"),
            "closing": first(tr, "dateValue"),
            "location": first(tr, "regionValue"),
            "published": first(tr, "publicationDate").replace("Published Date", "").strip(),
            "blurb": first(tr, "solicitationDescription")[:300],
            "mandatory_prebid": "Mandatory Pre-Bid" in t,
            "url": base_url + f"/private/solicitations/{sid}/abstract",
        })
    return rows


def result_count(d):
    body = text(d.body)
    m = re.search(r"(\d+)\s*-\s*(\d+)\s*of\s*(\d+)\s*results", body)
    if m:
        return (int(m.group(1)), int(m.group(2)), int(m.group(3)))
    m = re.search(r"(\d+)\s*Total Bids", body)
    return (1, 0, int(m.group(1))) if m else None


def saved_searches(d) -> list[tuple[str, str]]:
    out = []
    for o in d.xpath("//select[@id='selectedSavedSearch']/option"):
        val = (o.get("value") or "").strip()
        name = text(o)
        if val and val not in ("", "0") and not name.startswith("-"):
            out.append((name, val))
    return out


def parse_abstract(d):
    body = text(d.body)
    title = text(d.xpath("//h1")[0]) if d.xpath("//h1") else (d.findtext(".//title") or "")
    fields = {}
    for label in ("Solicitation Number", "Reference Number", "Solicitation Type", "Organization",
                  "Purchasing Group", "Publication Date", "Closing Date", "Closing Time", "Location",
                  "Bid Bond", "Pre-Bid", "Pre-Proposal", "Contact", "Email", "Phone", "Delivery",
                  "Piggyback", "Category"):
        m = re.search(re.escape(label) + r"\s*:?\s*(.{1,160}?)(?=\s(?:[A-Z][a-z]+ ?){1,3}\s*:|$)", body)
        if m:
            fields[label] = m.group(1).strip()
    dm = re.search(r"Description\s*:?\s*(.*?)(?:Documents|Attachments|Related Solicitations|Terms & Conditions)",
                   body, re.S)
    description = dm.group(1).strip() if dm else ""
    return title, fields, description, body


class BidNetSource:
    name = "bidnet"

    def __init__(self, cfg: dict):
        self.cfg = cfg
        self.base = cfg.get("base_url", "https://www.bidnetdirect.com").rstrip("/")
        self.page_size = int(cfg.get("page_size", 100))
        self.max_pages = int(cfg.get("max_pages", 60))
        self.pause = float(cfg.get("pause_seconds", 1.0))
        self.saved_search_ids = [str(x) for x in (cfg.get("saved_search_ids") or [])]
        self._session = None
        self.log = print

    # ---------------------------------------------------------------- http
    def _s(self) -> requests.Session:
        if self._session is None:
            try:
                cookie = cookie_header_from_curl_file(self.cfg.get("cookie_file", "~/.bidnet-curl.txt"))
            except (CookieFileMissing, CookieNotFound) as e:
                raise NeedsCookie(self.name, str(e)) from None
            s = requests.Session()
            s.headers.update({"User-Agent": UA, "Accept": "text/html,application/xhtml+xml,*/*;q=0.8",
                              "Accept-Language": "en-US,en;q=0.9", "Cookie": cookie})
            self._session = s
        return self._session

    def get(self, path: str) -> requests.Response:
        url = path if path.startswith("http") else self.base + path
        r = None
        for attempt in range(3):
            try:
                r = self._s().get(url, timeout=90, allow_redirects=True)
            except requests.RequestException:
                if attempt == 2:
                    raise
                time.sleep(2 + attempt * 2)
                continue
            if r.status_code >= 500:
                time.sleep(2 + attempt * 2)
                continue
            break
        time.sleep(self.pause)
        if "/public/authentication/login" in r.url or r.url.split("?")[0].endswith("/login"):
            raise NeedsCookie(self.name, f"redirected to login; re-paste {self.cfg.get('cookie_file')}")
        r.raise_for_status()
        return r

    def doc(self, r: requests.Response):
        return strip_scripts(lxhtml.fromstring(r.content, base_url=r.url))

    # ---------------------------------------------------------- protocol
    def healthcheck(self) -> str:
        try:
            d = self.doc(self.get("/private/supplier/solicitations/search"))
        except NeedsCookie:
            return HEALTH_NEEDS_COOKIE
        except requests.RequestException:
            return HEALTH_DOWN
        return HEALTH_OK if saved_searches(d) or result_count(d) else HEALTH_NEEDS_COOKIE

    def _searches(self) -> list[tuple[str, str]]:
        d = self.doc(self.get("/private/supplier/solicitations/search"))
        found = saved_searches(d)
        if self.saved_search_ids:
            by_id = dict((v, n) for n, v in found)
            return [(by_id.get(i, i), i) for i in self.saved_search_ids]
        if not found:
            raise NeedsCookie(self.name, "no saved searches on the search page; is the session live?")
        return found

    def run_search(self, name: str, val: str) -> tuple[list[dict], int | None]:
        rows, seen, total = [], set(), None
        for page in range(1, self.max_pages + 1):
            d = self.doc(self.get(f"/private/supplier/solicitations/search?pageNumber={page}"
                                  f"&pageSize={self.page_size}&selectedSavedSearch={val}"))
            rc = result_count(d)
            if rc:
                total = rc[2]
            page_rows = parse_rows(d, self.base)
            new = 0
            for row in page_rows:
                if row["id"] not in seen:
                    seen.add(row["id"])
                    rows.append(row)
                    new += 1
            if new == 0 or len(page_rows) < self.page_size or (rc and rc[1] >= rc[2]):
                break
        self.log(f"  bidnet/{name}: {len(rows)} rows (site says {total})")
        return rows, total

    def sweep(self) -> Iterator[dict]:
        allrows: dict[str, dict] = {}
        for name, val in self._searches():
            rows, _ = self.run_search(name, val)
            for row in rows:
                if row["id"] in allrows:
                    allrows[row["id"]]["searches"].append(name)
                else:
                    row["searches"] = [name]
                    allrows[row["id"]] = row
        yield from allrows.values()

    # ---------------------------------------------------------- abstract
    def abstract(self, sid: str) -> tuple[str, dict, str]:
        r = self.get(f"/private/supplier/interception/open-solicitation/{sid}?target=view")
        base = r.url.split("?")[0]
        for candidate in (base, f"{self.base}/private/supplier/solicitations/notice/{sid}/abstract"):
            try:
                r2 = self.get(candidate)
            except requests.HTTPError as e:
                if e.response is not None and e.response.status_code == 404:
                    continue
                raise
            if re.search(r"<title>\s*No javascript", r2.text, re.I):
                raise FederalNotice(f"{sid}: SAML page returned for {candidate}; session may need a re-paste")
            title, fields, description, _ = parse_abstract(self.doc(r2))
            return candidate, {"title": title, **fields}, description
        raise FederalNotice(f"{sid}: 404 on both abstract paths; federal notices need SAM.gov")
```

```python
"""SAM.gov Opportunities API v2. Public data; needs a free api.data.gov key in SAM_API_KEY.
Docs: https://open.gsa.gov/api/get-opportunities-public-api/
Search 'design-build', never 'modular' alone: federal notices are written that way."""
from __future__ import annotations

import datetime as dt
import time
from typing import Iterator

import requests

from .base import NeedsKey, HEALTH_OK, HEALTH_NEEDS_KEY, HEALTH_DOWN

DEFAULT_URL = "https://api.sam.gov/opportunities/v2/search"


def build_params(api_key: str, keyword: str | None, naics: str | None, ptypes: list[str],
                 posted_from: dt.date, posted_to: dt.date, limit: int, offset: int) -> dict:
    p = {"api_key": api_key, "postedFrom": posted_from.strftime("%m/%d/%Y"),
         "postedTo": posted_to.strftime("%m/%d/%Y"), "ptype": ",".join(ptypes),
         "limit": str(limit), "offset": str(offset)}
    if keyword:
        p["title"] = keyword
    if naics:
        p["ncode"] = naics
    return p


def parse_search(payload: dict) -> list[dict]:
    return list(payload.get("opportunitiesData") or [])


class SamGovSource:
    name = "samgov"

    def __init__(self, cfg: dict, last_run: dt.datetime | dt.date | None):
        self.cfg = cfg
        self.url = cfg.get("base_url", DEFAULT_URL)
        self.api_key = (cfg.get("api_key") or "").strip()
        self.keywords = list(cfg.get("keywords") or [])
        self.naics = [str(n) for n in (cfg.get("naics") or [])]
        self.ptypes = list(cfg.get("ptypes") or ["o", "p", "k"])
        self.lookback_days = int(cfg.get("lookback_days", 7))
        self.limit = int(cfg.get("limit", 1000))
        self.last_run = last_run.date() if isinstance(last_run, dt.datetime) else last_run
        self.log = print

    def healthcheck(self) -> str:
        if not self.api_key:
            return HEALTH_NEEDS_KEY
        try:
            r = requests.get(self.url, params=build_params(self.api_key, "housing", None, ["o"],
                                                            dt.date.today() - dt.timedelta(days=1),
                                                            dt.date.today(), 1, 0), timeout=60)
        except requests.RequestException:
            return HEALTH_DOWN
        return HEALTH_OK if r.status_code == 200 else (HEALTH_NEEDS_KEY if r.status_code in (401, 403) else HEALTH_DOWN)

    def _window(self) -> tuple[dt.date, dt.date]:
        today = dt.date.today()
        start = (self.last_run - dt.timedelta(days=1)) if self.last_run else today - dt.timedelta(days=self.lookback_days)
        return start, today

    def _page(self, keyword: str | None, naics: str | None, offset: int) -> dict:
        start, end = self._window()
        params = build_params(self.api_key, keyword, naics, self.ptypes, start, end, self.limit, offset)
        for attempt in range(3):
            r = requests.get(self.url, params=params, timeout=90)
            if r.status_code == 429 or r.status_code >= 500:
                time.sleep(5 + attempt * 5)
                continue
            if r.status_code in (401, 403):
                raise NeedsKey(self.name, f"HTTP {r.status_code} from SAM.gov")
            r.raise_for_status()
            return r.json()
        r.raise_for_status()
        return {}

    def sweep(self) -> Iterator[dict]:
        if not self.api_key:
            raise NeedsKey(self.name, "set SAM_API_KEY")
        seen: set[str] = set()
        queries = [(k, None) for k in self.keywords] + [(None, n) for n in self.naics]
        for keyword, naics in queries:
            offset, got = 0, 0
            while True:
                payload = self._page(keyword, naics, offset)
                recs = parse_search(payload)
                for rec in recs:
                    nid = rec.get("noticeId")
                    if nid and nid not in seen:
                        seen.add(nid)
                        got += 1
                        yield rec
                total = int(payload.get("totalRecords") or 0)
                offset += len(recs)
                if not recs or offset >= total:
                    break
                time.sleep(1.0)
            self.log(f"  samgov/{keyword or naics}: {got} new records")
```

```python
"""PublicPurchase.com adapter. Verified 2026-09-04: region -> agency list is a plain GET
(/gems/global/home/getAgenciesByRegion?region=XX); each agency's list is /gems/<slug>/global/home/nationalBidList.
Without a vendor cookie every agency answers "No bids at this time", so this adapter reports needs_cookie
and yields nothing rather than pretending the market is empty."""
from __future__ import annotations

import re
import time
from typing import Iterator

import requests
from lxml import html as lxhtml

from ..cookies import cookie_header_from_curl_file, CookieFileMissing, CookieNotFound
from .base import HEALTH_OK, HEALTH_NEEDS_COOKIE, HEALTH_DOWN

UA = ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
      "(KHTML, like Gecko) Chrome/124.0 Safari/537.36")


def parse_agencies(payload, agency_pick: re.Pattern) -> list[dict]:
    out = []
    for a in payload or []:
        name = a.get("agencyName") or a.get("name") or ""
        slug = a.get("agencyId") or a.get("id") or ""
        if slug and agency_pick.search(name):
            out.append({"slug": str(slug), "name": name, "state": a.get("state", "")})
    return out


def parse_bid_list(html_text: str, agency: dict, base_url: str) -> list[dict]:
    d = lxhtml.fromstring(html_text)
    rows = []
    for tr in d.xpath("//tr[contains(@class,'bidRow')]|//tr[.//a[contains(@href,'bidId=')]]"):
        a = tr.xpath(".//a[contains(@href,'bidId=')]")
        if not a:
            continue
        a = a[0]
        m = re.search(r"bidId=(\d+)", a.get("href") or "")
        if not m:
            continue
        t = re.sub(r"\s+", " ", tr.text_content()).strip()
        pub = re.search(r"Published\s*([\d/]+)", t)
        clo = re.search(r"Closes?\s*([\d/]+(?:\s+[\d:]+\s*[AP]M(?:\s*[A-Z]{2,4})?)?)", t)
        rows.append({
            "id": m.group(1), "title": re.sub(r"\s+", " ", a.text_content()).strip(),
            "agency": agency["name"], "agency_slug": agency["slug"], "location": agency.get("state", ""),
            "published": pub.group(1) if pub else "", "closing": clo.group(1).strip() if clo else "",
            "url": base_url + a.get("href"), "blurb": "",
        })
    return rows


class PublicPurchaseSource:
    name = "publicpurchase"

    def __init__(self, cfg: dict):
        self.cfg = cfg
        self.base = cfg.get("base_url", "https://www.publicpurchase.com").rstrip("/")
        self.regions = list(cfg.get("regions") or ["CO"])
        self.agency_pick = re.compile(cfg.get("agency_pick") or "hous", re.I)
        self.bid_pick = re.compile(cfg.get("bid_pick") or "housing", re.I)
        self.pause = float(cfg.get("pause_seconds", 1.0))
        self._session = None
        self.log = print

    def _s(self) -> requests.Session | None:
        if self._session is None:
            try:
                cookie = cookie_header_from_curl_file(self.cfg.get("cookie_file", "~/.publicpurchase-curl.txt"))
            except (CookieFileMissing, CookieNotFound):
                return None
            s = requests.Session()
            s.headers.update({"User-Agent": UA, "Cookie": cookie, "Accept": "*/*"})
            self._session = s
        return self._session

    def healthcheck(self) -> str:
        s = self._s()
        if s is None:
            return HEALTH_NEEDS_COOKIE
        try:
            r = s.get(f"{self.base}/gems/global/home/getAgenciesByRegion", params={"region": self.regions[0]}, timeout=60)
        except requests.RequestException:
            return HEALTH_DOWN
        if r.status_code != 200 or "login" in r.url:
            return HEALTH_NEEDS_COOKIE
        return HEALTH_OK

    def sweep(self) -> Iterator[dict]:
        s = self._s()
        if s is None:
            self.log("  publicpurchase: no vendor cookie; skipping")
            return
        for region in self.regions:
            r = s.get(f"{self.base}/gems/global/home/getAgenciesByRegion", params={"region": region}, timeout=60)
            if r.status_code != 200:
                continue
            try:
                payload = r.json()
            except ValueError:
                continue
            for agency in parse_agencies(payload, self.agency_pick):
                rb = s.get(f"{self.base}/gems/{agency['slug']}/global/home/nationalBidList", timeout=60)
                time.sleep(self.pause)
                if rb.status_code != 200 or "No bids at this time" in rb.text:
                    continue
                for row in parse_bid_list(rb.text, agency, self.base):
                    if self.bid_pick.search(row["title"]):
                        yield row
```

```python
"""bid-radar command line."""
from __future__ import annotations

import argparse
import datetime as dt
import sys
import traceback
from pathlib import Path

from .config import load_config, resolve_path
from .cookies import redact
from .dedupe import link_duplicates
from .digest import write_digest
from .normalize import normalize
from .scoring import Rules, score
from .sources.base import NeedsCookie, NeedsKey
from .sources.bidnet import BidNetSource, FederalNotice
from .sources.publicpurchase import PublicPurchaseSource
from .sources.samgov import SamGovSource
from .store import Store


def _store(cfg: dict) -> Store:
    return Store(resolve_path(cfg, "data_dir") / "bid-radar.sqlite")


def build_sources(cfg: dict, store: Store, only: list[str] | None = None) -> list:
    sc = cfg["sources"]
    out = []
    if sc["bidnet"].get("enabled", True):
        out.append(BidNetSource(sc["bidnet"]))
    if sc["samgov"].get("enabled", True):
        out.append(SamGovSource(sc["samgov"], last_run=store.last_successful_run()))
    if sc["publicpurchase"].get("enabled", True):
        out.append(PublicPurchaseSource(sc["publicpurchase"]))
    if only:
        out = [s for s in out if s.name in only]
    return out


def run_sweep(cfg: dict, sources: list | None = None, only: list[str] | None = None,
              now: dt.datetime | None = None) -> dict:
    now = now or dt.datetime.now()
    store = _store(cfg)
    rules = Rules.from_config(cfg["scoring"])
    sources = sources if sources is not None else build_sources(cfg, store, only)
    run_id = store.start_run()
    summary: dict[str, dict] = {}
    errors = []
    for src in sources:
        rows, new = 0, 0
        try:
            for raw in src.sweep():
                sol = normalize(src.name, raw)
                s, note = score(sol, rules)
                if store.upsert(sol, s, note, now):
                    new += 1
                rows += 1
            summary[src.name] = {"rows": rows, "new": new, "health": "ok"}
        except NeedsCookie as e:
            summary[src.name] = {"rows": rows, "new": new, "health": "needs_cookie"}
            errors.append(redact(str(e)))
        except NeedsKey as e:
            summary[src.name] = {"rows": rows, "new": new, "health": "needs_key"}
            errors.append(str(e))
        except Exception as e:  # noqa: BLE001 - a source failing must not stop the sweep
            summary[src.name] = {"rows": rows, "new": new, "health": "down"}
            errors.append(f"{src.name}: {redact(repr(e))}")
            traceback.print_exc()
        print(f"{src.name}: {summary[src.name]}")
    links = link_duplicates(store, store.all_rows())
    print(f"duplicate links added: {links}")
    store.finish_run(run_id, summary, "; ".join(errors) if errors else None)
    store.close()
    return summary


def cmd_whoami(cfg, args):
    store = _store(cfg)
    for src in build_sources(cfg, store, args.only):
        print(f"{src.name}: {src.healthcheck()}")
    return 0


def cmd_sweep(cfg, args):
    run_sweep(cfg, only=args.only)
    return cmd_digest(cfg, args)


def cmd_rescore(cfg, args):
    store = _store(cfg)
    rules = Rules.from_config(cfg["scoring"])
    changed = 0
    for row in store.all_rows():
        sol = store.get(row["uid"])
        s, note = score(sol, rules)
        if s != row["score"] or note != row["score_note"]:
            store.db.execute("update solicitations set score=?, score_note=? where uid=?", (s, note, row["uid"]))
            changed += 1
    store.db.commit()
    print(f"rescored: {changed} rows changed")
    return 0


def cmd_digest(cfg, args):
    store = _store(cfg)
    rules = Rules.from_config(cfg["scoring"])
    since = dt.datetime.fromisoformat(args.since) if getattr(args, "since", None) else _previous_run_start(store)
    md, js = write_digest(store, rules, resolve_path(cfg, "output_dir"), dt.datetime.now(), since)
    print(f"digest: {md}\n        {js}")
    return 0


def _previous_run_start(store: Store) -> dt.datetime | None:
    rows = store.db.execute("select started from runs where finished is not null order by id desc limit 2").fetchall()
    return dt.datetime.fromisoformat(rows[1]["started"]) if len(rows) > 1 else None


def cmd_abstract(cfg, args):
    src = BidNetSource(cfg["sources"]["bidnet"])
    for sid in args.ids:
        try:
            url, fields, description = src.abstract(sid)
        except FederalNotice as e:
            print(f"! {e}")
            continue
        print(f"== {sid}  {url}")
        for k, v in fields.items():
            print(f"  {k}: {v}")
        print("  Description:", description[:1500])
    return 0


def cmd_mark(cfg, args):
    _store(cfg).mark(args.uid, args.status)
    print(f"{args.uid} -> {args.status}")
    return 0


def main(argv=None) -> int:
    p = argparse.ArgumentParser(prog="bid-radar")
```
